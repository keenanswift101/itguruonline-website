    const ADMIN_REMINDER_EMAIL = "ambrose@it-guru.co.za"

    export async function runEnquiryReminderJob(
      opts?: { triggeredBy?: string; now?: Date }
    ): Promise<{ sent: number; skipped: number }> {
      const now = opts?.now ?? new Date()
      const triggeredBy = opts?.triggeredBy ?? "scheduled"
      const today = now.toISOString().slice(0, 10) // YYYY-MM-DD

      try {
        // 1. Read enquiry_stale_days from site_settings
        const [setting] = await db
          .select({ value: siteSettings.value })
          .from(siteSettings)
          .where(eq(siteSettings.key, "enquiry_stale_days"))
        const staleDays = parseInt(setting?.value ?? "7", 10)

        // 2. Query stale enquiries not yet reminded today
        // stale = status != 'completed' AND updated_at < NOW() - INTERVAL N days
        // not reminded = last_reminded_at IS NULL OR last_reminded_at < today
        const cutoffDate = new Date(now.getTime() - staleDays * 24 * 60 * 60 * 1000)
        const staleEnquiries = await db
          .select({
            id: contactEnquiries.id,
            name: contactEnquiries.name,
            updatedAt: contactEnquiries.updatedAt,
            lastRemindedAt: contactEnquiries.lastRemindedAt,
          })
          .from(contactEnquiries)
          .where(
            and(
              ne(contactEnquiries.status, "completed"),
              lt(contactEnquiries.updatedAt, cutoffDate),
              or(
                isNull(contactEnquiries.lastRemindedAt),
                lt(contactEnquiries.lastRemindedAt, sql`CURRENT_DATE`)
              )
            )
          )

        // 3. Count already-reminded today (for skipped count — query separately)
        const alreadyRemindedToday = await db
          .select({ id: contactEnquiries.id })
          .from(contactEnquiries)
          .where(
            and(
              ne(contactEnquiries.status, "completed"),
              lt(contactEnquiries.updatedAt, cutoffDate),
              eq(contactEnquiries.lastRemindedAt, sql`CURRENT_DATE`)
            )
          )

        const skipped = alreadyRemindedToday.length

        if (staleEnquiries.length === 0) {
          // Write run record — nothing to do
          await db.insert(automationRuns).values({
            jobName: "enquiry-reminder",
            ranAt: now,
            triggeredBy,
            status: "success",
            resultSummary: "0 stale enquiries",
          })
          return { sent: 0, skipped }
        }

        // 4. Build email body — table-based inline styles (no flexbox/grid per CLAUDE.md)
        const rows = staleEnquiries
          .map((e) => {
            const daysSince = Math.floor(
              (now.getTime() - new Date(e.updatedAt).getTime()) / (1000 * 60 * 60 * 24)
            )
            return `<tr>
              <td style="padding:8px 12px; border-bottom:1px solid #e2e8f0;">${escapeHtml(e.name)}</td>
              <td style="padding:8px 12px; border-bottom:1px solid #e2e8f0; color:#ef4444;">${daysSince} days</td>
            </tr>`
          })
          .join("")

        const bodyHtml = `
          <p style="margin:0 0 16px 0;">The following ${staleEnquiries.length} enquiry(ies) have had no status change in over ${staleDays} days:</p>
          <table role="presentation" width="100%" cellpadding="0" cellspacing="0" border="0" style="border-collapse:collapse; border:1px solid #e2e8f0;">
            <tr>
              <th style="padding:8px 12px; background:#f8fafc; text-align:left; font-size:12px; text-transform:uppercase; letter-spacing:0.05em;">Name</th>
              <th style="padding:8px 12px; background:#f8fafc; text-align:left; font-size:12px; text-transform:uppercase; letter-spacing:0.05em;">Stale For</th>
            </tr>
            ${rows}
          </table>
          <p style="margin:16px 0 0 0;">Please follow up or update their status in the admin panel.</p>
        `

        const subject = `Stale Enquiry Reminder — ${staleEnquiries.length} enquiry(ies) need attention`
        await sendEmail({
          to: ADMIN_REMINDER_EMAIL,
          subject,
          html: emailLayout(subject, bodyHtml),
        })

        // 5. Update last_reminded_at for all reminded records
        const remindedIds = staleEnquiries.map((e) => e.id)
        await db
          .update(contactEnquiries)
          .set({ lastRemindedAt: today })
          .where(inArray(contactEnquiries.id, remindedIds))

        // 6. Write automation_runs row
        const summary = `Sent reminder for ${staleEnquiries.length} stale enquiry(ies)`
        await db.insert(automationRuns).values({
          jobName: "enquiry-reminder",
          ranAt: now,
          triggeredBy,
          status: "success",
          resultSummary: summary,
        })

        return { sent: 1, skipped }
      } catch (err) {
        await db.insert(automationRuns).values({
          jobName: "enquiry-reminder",
          ranAt: now,
          triggeredBy,
          status: "error",
          errorMessage: String(err),
        })
        throw err
      }
    }

          .where(eq(siteSettings.key, "invoice_overdue_reminder_days"))

    import { db } from "@/lib/db/index"
    import { billingSchedules, hostingPackages, invoices, invoiceLineItems, automationRuns } from "@/lib/db/schema"
    import { eq, and } from "drizzle-orm"

    export async function runRecurringBillingJob(
      opts?: { triggeredBy?: string; now?: Date }
    ): Promise<{ inserted: number; skipped: number }> {
      const now = opts?.now ?? new Date()
      const triggeredBy = opts?.triggeredBy ?? "scheduled"

      // billingPeriodStart = 1st of current month (per D-03 and RESEARCH.md Section 6)
      const billingPeriodStart = new Date(now.getFullYear(), now.getMonth(), 1)
      const billingPeriodStartStr = billingPeriodStart.toISOString().slice(0, 10)

      // issueDate = today, dueDate = 30 days from today
      const issueDateStr = now.toISOString().slice(0, 10)
      const dueDateObj = new Date(now.getTime() + 30 * 24 * 60 * 60 * 1000)
      const dueDateStr = dueDateObj.toISOString().slice(0, 10)

      try {
        // 1. Query active billing schedules joined with their hosting package
        const activeSchedules = await db
          .select({
            scheduleId: billingSchedules.id,
            clientName: billingSchedules.clientName,
            clientEmail: billingSchedules.clientEmail,
            packageId: billingSchedules.packageId,
            packageName: hostingPackages.name,
            packagePriceRands: hostingPackages.priceRands,
          })
          .from(billingSchedules)
          .leftJoin(hostingPackages, eq(billingSchedules.packageId, hostingPackages.id))
          .where(eq(billingSchedules.isActive, true))

        if (activeSchedules.length === 0) {
          await db.insert(automationRuns).values({
            jobName: "recurring-billing",
            ranAt: now,
            triggeredBy,
            status: "success",
            resultSummary: "0 active billing schedules",
          })
          return { inserted: 0, skipped: 0 }
        }

        let insertedCount = 0
        let skippedCount = 0

        // 2. For each active schedule, attempt invoice insert with ON CONFLICT DO NOTHING
        for (const schedule of activeSchedules) {
          const priceRands = schedule.packagePriceRands ?? 0
          const packageName = schedule.packageName ?? "Hosting Package"

          // Insert invoice with ON CONFLICT DO NOTHING on (billingScheduleId, billingPeriodStart)
          const result = await db
            .insert(invoices)
            .values({
              clientName: schedule.clientName,
              clientEmail: schedule.clientEmail ?? null,
              status: "draft",
              issueDate: issueDateStr,
              dueDate: dueDateStr,
              totalRands: priceRands,
              billingScheduleId: schedule.scheduleId,
              billingPeriodStart: billingPeriodStartStr,
            })
            .onConflictDoNothing()
            .returning({ id: invoices.id })

          if (result.length === 0) {
            // Conflict — invoice already exists for this schedule + period
            skippedCount++
            continue
          }

          // Invoice was inserted — add line item
          const newInvoiceId = result[0].id
          await db.insert(invoiceLineItems).values({
            invoiceId: newInvoiceId,
            description: `${packageName} – monthly hosting`,
            quantity: 1,
            unitPriceRands: priceRands,
            lineTotalRands: priceRands,
            sortOrder: 1,
          })

          insertedCount++
        }

        // 3. Write automation_runs row
        const summary = `Inserted ${insertedCount} invoice(s), skipped ${skippedCount} (already exists)`
        await db.insert(automationRuns).values({
          jobName: "recurring-billing",
          ranAt: now,
          triggeredBy,
          status: "success",
          resultSummary: summary,
        })

        return { inserted: insertedCount, skipped: skippedCount }
      } catch (err) {
        await db.insert(automationRuns).values({
          jobName: "recurring-billing",
          ranAt: now,
          triggeredBy,
          status: "error",
          errorMessage: String(err),
        })
        throw err
      }
    }
