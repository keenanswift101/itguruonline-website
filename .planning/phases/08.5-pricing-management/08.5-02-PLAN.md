    ```typescript
    const CreateSchema = z.object({
      name: z.string().trim().min(1).max(64),
      priceRands: z.number().int().positive(),
      pricePeriod: z.enum(["mo", "yr"]).default("mo"),
      description: z.string().max(2000).optional().default(""),
      features: z.string().max(4000).optional().default(""), // newline-separated, stored raw
      isPopular: z.boolean().optional().default(false),
    });

    export async function POST(req: NextRequest) {
      const session = await requireAdmin();
      if (!session) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });

      let raw: unknown;
      try { raw = await req.json(); } catch { return NextResponse.json({ error: "Invalid request body." }, { status: 400 }); }
      const parsed = CreateSchema.safeParse(raw);
      if (!parsed.success) {
        return NextResponse.json({ error: "Validation failed.", fields: parsed.error.flatten().fieldErrors }, { status: 422 });
      }
      // Derive slug from name (prevents typo slugs breaking ?package=slug matching)
      const slug = parsed.data.name.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-+|-+$/g, "");
      if (!slug) return NextResponse.json({ error: "Validation failed.", fields: { name: ["Name must contain letters or numbers"] } }, { status: 422 });
      const [existing] = await db.select({ id: hostingPackages.id }).from(hostingPackages).where(eq(hostingPackages.slug, slug));
      if (existing) return NextResponse.json({ error: "Validation failed.", fields: { name: ["A package with a similar name already exists"] } }, { status: 422 });
      const [{ max }] = await db.select({ max: sql<number>`coalesce(max(${hostingPackages.sortOrder}), 0)` }).from(hostingPackages);
      const [created] = await db.insert(hostingPackages).values({
        slug, name: parsed.data.name, priceRands: parsed.data.priceRands, pricePeriod: parsed.data.pricePeriod,
        description: parsed.data.description, features: parsed.data.features, isPopular: parsed.data.isPopular,
        sortOrder: (max ?? 0) + 1, isActive: true,
      }).returning({ id: hostingPackages.id });
      return NextResponse.json({ id: created.id }, { status: 201 });
    }
    ```
